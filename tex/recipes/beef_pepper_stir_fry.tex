\recdef{Beef Pepper Stir-Fry}
Stir fried beef and shishito peppers.

\begin{rec}
\ing{250 g}{skirt steak, thinly sliced against the grain}
\ing{15 g}{soy sauce}
\ing{15 g}{mirin}
\ing{3 g}{\recref{Thai Kosho}}
Whisk together soy sauce, mirin, and Thai kosho.
Combine with sliced beef and marinate at room temperature for 30 minutes.
\ing{5 g}{potato starch}
\ing{5 g}{\recref{Trinity Oil}}
Add potato starch and toss to coat.
Drizzle with trinity oil and toss.
\ing{15 g}{\recref{Trinity Oil}}
\ing{150 g}{shishito peppers, whole, poked}
Heat cast iron pan over high heat until smoking.
Add trinity oil and blister shishito peppers.
Remove and set aside.
\ing{15 g}{\recref{Trinity Oil}}
Wipe pan and reheat until smoking.
Add trinity oil.
Sear beef in a single layer in batches and set aside.
\ing{60 g}{\recref{Chicken Stock}}
Pour off excess fat from pan and deglaze with chicken stock.
Reduce the stock to 30 g.
\ing{20 g}{\recref{Fermented Ginger Honey}}
\ing{10 g}{\recref{Frozen Lime Juice}, thawed}
\ing{5 g}{\recref{Cilantro Oil}}
Stir in honey and lime juice.
Return everything and stir until thickened.
Stir in cilantro oil and transfer to serving plate.
\end{rec}
