\recdef{Chicken Stock}
Basic pressure-cooked chicken stock.

\begin{rec}
\ing{1.5 kg}{chicken parts}
\ing{1.5 kg}{water}
\ing{300 g}{white onion, diced}
\ing{150 g}{carrots, diced}
\ing{150 g}{celery root, diced}
\ing{6 sprigs}{thyme}
\ing{3}{bay leaves}
\ing{9 g}{black peppercorns}
Combine in electric pressure cooker and pressure-cook for 1 hour.
\newstep
Naturally release pressure.
Strain and discard solids.
Cool to room temperature and refrigerate overnight.
\newstep
Remove the solidified fat layer.
Freeze into 30g portions.
\end{rec}
