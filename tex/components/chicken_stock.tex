\recdef{Chicken Stock}
Basic pressure-cooked chicken wing stock.

\begin{rec}
\ing{2 kg}{chicken wings}
\ing{2 kg}{water}
\ing{400 g}{white onion, diced}
\ing{200 g}{carrots, diced}
\ing{200 g}{celery root, diced}
\ing{8 sprigs}{thyme}
\ing{4}{bay leaves}
\ing{10 g}{black peppercorns}
Combine in electric pressure cooker and pressure-cook for 1 hour.
\newstep
Naturally release pressure. Strain and discard solids. Portion and store frozen.
\end{rec}
