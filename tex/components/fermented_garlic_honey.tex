\recdef{Fermented Garlic Honey}
Lactic acid ferment of garlic in honey.

\begin{rec}
\ing{200 g}{raw honey}
\ing{100 g}{Italian softneck garlic, peeled, lightly crushed}
Combine honey and garlic in sterilized self-burping jar.
Let ferment for 2-4 weeks at room temperature.
\newstep
Store refrigerated.
\end{rec}
