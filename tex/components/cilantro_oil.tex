\recdef{Cilantro Oil}
Fresh cilantro extracted in sunflower oil.

\begin{rec}
\ing{100 g}{cilantro (leaves + tender stems)}
\ing{300 g}{unrefined sunflower oil}
Blend together into a smooth puree.
Store refrigerated for 1 day.
\newstep
Strain oil and discard solids. Store in freezer.
\end{rec}
