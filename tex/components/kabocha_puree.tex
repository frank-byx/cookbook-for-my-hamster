\recdef{Kabocha Puree}
Kabocha puree without liquid.

\begin{rec}
\ing{1 medium}{kabocha squash, halved, deseeded}
Roast cut side down at 180$^\circ$C for 45-60 minutes until very soft.
\newstep
Pass flesh through fine mesh sieve. Portion and store frozen.
\end{rec}
