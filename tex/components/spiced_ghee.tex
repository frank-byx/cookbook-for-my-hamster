\recdef{Spiced Ghee}
Browned butter infused with spices.

\begin{rec}
\ing{3}{star anise}
\ing{8}{cloves}
\ing{1}{cassia stick}
\ing{1 tsp}{fennel seeds}
\ing{1 tsp}{Sichuan peppercorns}
\ing{1}{black cardamom pod, lightly crushed}
\ing{2}{dried bay leaves}
Gently toast spices individually.
\ing{1 kg}{unsalted butter}
Simmer butter until milk solids separate.
Add toasted spices to butter and hold at 120$^\circ$C until milk solids turn brown.
Let cool to room temperature.
\newstep
Strain oil and discard solids. Store in freezer.
\end{rec}
