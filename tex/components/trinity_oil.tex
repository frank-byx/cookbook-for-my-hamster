\recdef{Trinity Oil}
Extraction of ginger, scallion, and garlic in neutral oil.

\begin{rec}
\ing{300 g}{neutral oil}
Heat oil to 160$^\circ$C.
\ing{50 g}{scallions, whole, thinly sliced}
\ing{25 g}{fresh ginger, skin on, thinly sliced}
\ing{25 g}{garlic, germ removed, thinly sliced}
Simmer in oil until garlic turns golden. Cool to room temperature.
\newstep
Strain oil and discard solids. Store in freezer.
\end{rec}
