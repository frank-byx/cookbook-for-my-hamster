\recdef{Fermented Ginger Honey}
Lactic acid ferment of ginger in honey.

\begin{rec}
\ing{150 g}{ginger, skin-on, sliced}
Wash ginger with cold filtered water and pat dry.
\ing{500 g}{raw honey}
Combine honey and ginger in sterilized self-burping jar.
Let ferment for 2-4 weeks at room temperature.
\newstep
Store refrigerated.
\end{rec}
